# Libraries are split so the recompiler tool, the runtime and the AOT module DLLs can link only
# what they need. pspc_sem (decode + semantics) is the only thing an AOT DLL links.

add_library(pspc_common STATIC
	common/log.cpp
	common/fileutil.cpp
	common/sha1.cpp
)
target_include_directories(pspc_common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

file(GLOB PSPC_VFPU_EXTRA_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/allegrex/vfpu_*.cpp)
add_library(pspc_sem STATIC
	allegrex/decode.cpp
	allegrex/disasm.cpp
	allegrex/sem_tables.cpp
	${PSPC_VFPU_EXTRA_SOURCES}
)
target_include_directories(pspc_sem PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

# Semantics version: a hash of everything AOT-compiled code bakes in (src/allegrex). A DLL built
# from different semantics is stale and the runtime ignores it.
file(GLOB PSPC_SEM_FILES ${CMAKE_CURRENT_SOURCE_DIR}/allegrex/*)
list(SORT PSPC_SEM_FILES)
set(PSPC_SEM_HASH_INPUT "")
foreach(f ${PSPC_SEM_FILES})
	file(SHA1 ${f} h)
	string(APPEND PSPC_SEM_HASH_INPUT "${h}")
endforeach()
string(SHA1 PSPC_SEM_HASH "${PSPC_SEM_HASH_INPUT}")
string(SUBSTRING ${PSPC_SEM_HASH} 0 8 PSPC_SEM_HASH8)
file(WRITE ${CMAKE_BINARY_DIR}/generated/pspc_sem_version.h.tmp
	"// Generated by CMake from the contents of src/allegrex.\n#pragma once\n#define PSPC_SEM_VERSION 0x${PSPC_SEM_HASH8}u\n")
file(COPY_FILE ${CMAKE_BINARY_DIR}/generated/pspc_sem_version.h.tmp ${CMAKE_BINARY_DIR}/generated/pspc_sem_version.h ONLY_IF_DIFFERENT)
target_include_directories(pspc_sem PUBLIC ${CMAKE_BINARY_DIR}/generated)
target_link_libraries(pspc_sem PUBLIC pspc_common)

# Each component directory owns its CMakeLists.txt; order matters only for dependencies.
foreach(component fs loader gpu audio media hle runtime recomp tools)
	if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${component}/CMakeLists.txt)
		add_subdirectory(${component})
	endif()
endforeach()
