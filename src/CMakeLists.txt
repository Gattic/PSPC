# Libraries are split so the recompiler tool, the runtime and the AOT module DLLs can link only
# what they need. pspc_sem (decode + semantics) is the only thing an AOT DLL links.

add_library(pspc_common STATIC
	common/log.cpp
	common/fileutil.cpp
	common/app_paths.cpp
	common/firmware.cpp
	common/host_file.cpp
	common/host.cpp
	common/json.cpp
	common/seplugins.cpp
	common/settings.cpp
	common/sha1.cpp
)
target_include_directories(pspc_common PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(pspc_common PUBLIC Threads::Threads ${CMAKE_DL_LIBS})

file(GLOB PSPC_VFPU_EXTRA_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/allegrex/vfpu_*.cpp)
add_library(pspc_sem STATIC
	allegrex/decode.cpp
	allegrex/disasm.cpp
	allegrex/sem_tables.cpp
	${PSPC_VFPU_EXTRA_SOURCES}
)
target_include_directories(pspc_sem PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

# Semantics version includes the emitter and both routes' compiler policy, so cached modules
# cannot survive a code-generation or floating-point compilation fix. Keep source_sem_version()
# in tools/pspc_build.py in sync with these inputs.
file(GLOB PSPC_SEM_FILES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/allegrex/*)
list(APPEND PSPC_SEM_FILES ${CMAKE_CURRENT_SOURCE_DIR}/recomp/codegen.cpp ${PROJECT_SOURCE_DIR}/tools/pspc_build.py)
list(SORT PSPC_SEM_FILES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PSPC_SEM_FILES})
set(PSPC_SEM_HASH_INPUT "")
foreach(f ${PSPC_SEM_FILES})
	file(SHA1 ${f} h)
	string(APPEND PSPC_SEM_HASH_INPUT "${h}")
endforeach()
string(SHA1 PSPC_SEM_HASH "${PSPC_SEM_HASH_INPUT}")
string(SUBSTRING ${PSPC_SEM_HASH} 0 8 PSPC_SEM_HASH8)
file(WRITE ${CMAKE_BINARY_DIR}/generated/pspc_sem_version.h.tmp
	"// Generated by CMake from AOT semantics, codegen and compiler policy.\n#pragma once\n#define PSPC_SEM_VERSION 0x${PSPC_SEM_HASH8}u\n")
file(COPY_FILE ${CMAKE_BINARY_DIR}/generated/pspc_sem_version.h.tmp ${CMAKE_BINARY_DIR}/generated/pspc_sem_version.h ONLY_IF_DIFFERENT)
target_include_directories(pspc_sem PUBLIC ${CMAKE_BINARY_DIR}/generated)
target_link_libraries(pspc_sem PUBLIC pspc_common)

# Each component directory owns its CMakeLists.txt; order matters only for dependencies.
set(PSPC_COMPONENTS fs loader recomp)
if(PSPC_BUILD_RUNTIME)
	list(APPEND PSPC_COMPONENTS gpu audio media net runtime)
endif()
foreach(component ${PSPC_COMPONENTS})
	if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${component}/CMakeLists.txt)
		add_subdirectory(${component})
	endif()
endforeach()

# Disc extraction is needed by the AOT SDK even in a tools-only build.
add_executable(pspc-iso frontend/iso_tool.cpp)
target_link_libraries(pspc-iso PRIVATE pspc_fs pspc_common)

add_library(pspc_game_library STATIC frontend/game_library.cpp frontend/native_cache.cpp)
target_link_libraries(pspc_game_library PUBLIC pspc_fs)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
	target_sources(pspc_game_library PRIVATE frontend/child_process.cpp)
endif()
